<script>
(function () {
  var search = document.getElementById('baby-name-search');
  var gender = document.getElementById('baby-name-gender');
  var religion = document.getElementById('baby-name-religion');
  var letter = document.getElementById('baby-name-letter');
  var list = document.getElementById('baby-name-results');
  var items = Array.prototype.slice.call(list.querySelectorAll('li'));
  var count = document.getElementById('baby-name-result-count');
  var empty = document.getElementById('baby-name-no-results');
  var reset = document.getElementById('baby-name-reset');
  function applyFilters() {
    var q = search.value.trim().toLocaleLowerCase();
    var shown = 0;
    items.forEach(function (item) {
      var matches = (!q || (item.dataset.name + ' ' + item.dataset.meaning + ' ' + item.textContent).toLocaleLowerCase().indexOf(q) !== -1)
        && (gender.value === 'all' || item.dataset.gender === gender.value)
        && (religion.value === 'all' || item.dataset.religion === religion.value)
        && (letter.value === 'all' || item.dataset.letter === letter.value);
      item.hidden = !matches;
      if (matches) shown++;
    });
    count.textContent = shown + (shown === 1 ? ' name' : ' names') + ' found';
    empty.hidden = shown !== 0;
    list.hidden = shown === 0;
  }
  [search, gender, religion, letter].forEach(function (control) {
    control.addEventListener(control === search ? 'input' : 'change', applyFilters);
  });
  reset.addEventListener('click', function () {
    search.value = '';
    gender.value = 'all';
    religion.value = 'all';
    letter.value = 'all';
    applyFilters();
    search.focus();
  });
  applyFilters();
})();
</script>
